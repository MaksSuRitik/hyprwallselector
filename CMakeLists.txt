cmake_minimum_required(VERSION 3.25)
project(hyprwallselector VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Qml DBus Network Concurrent ShaderTools Test)
find_package(LayerShellQt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
add_library(hws-core STATIC src/controller.cpp src/controller.h src/mediaitem.cpp src/mediaitem.h src/integration.cpp src/integration.h)
target_include_directories(hws-core PUBLIC src)
qt_add_executable(hyprwallselector src/main.cpp)
target_link_libraries(hyprwallselector PRIVATE hws-core)
target_link_libraries(hws-core PUBLIC Qt6::Quick Qt6::QuickControls2 Qt6::DBus Qt6::Network Qt6::Concurrent LayerShellQt::Interface PkgConfig::MPV PkgConfig::SYSTEMD)
qt_add_resources(hws-core ui PREFIX / FILES qml/Main.qml qml/Card.qml qml/Settings.qml qml/Wallpaper.qml qml/SoftButton.qml)
qt_add_shaders(hws-core shaders PREFIX / FILES shaders/card.frag)
add_executable(hyprwallselector-hotkeys src/hotkeys.cpp src/chord.h)
target_link_libraries(hyprwallselector-hotkeys PRIVATE Qt6::Core Qt6::DBus PkgConfig::SYSTEMD PkgConfig::UDEV)
target_compile_options(hws-core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(hyprwallselector-hotkeys PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS hyprwallselector RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS hyprwallselector-hotkeys RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
install(FILES packaging/org.hyprwallselector.Hotkeys.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/system.d)
configure_file(packaging/org.hyprwallselector.Hotkeys.service.in org.hyprwallselector.Hotkeys.service @ONLY)
configure_file(packaging/hyprwallselector-hotkeys.service.in hyprwallselector-hotkeys.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.hyprwallselector.Hotkeys.service DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/system-services)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/hyprwallselector-hotkeys.service DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/system)
install(FILES packaging/hyprwallselector.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/hyprwallselector.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/hyprwallselector)
install(FILES docs/SECURITY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/hyprwallselector)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/hyprwallselector)
include(CTest)
if(BUILD_TESTING)
 add_executable(test-chord tests/test-chord.cpp src/chord.h)
 target_include_directories(test-chord PRIVATE src)
 target_link_libraries(test-chord PRIVATE Qt6::Test)
 add_test(NAME chord COMMAND test-chord)
 add_executable(test-ui tests/test-ui.cpp)
 target_link_libraries(test-ui PRIVATE hws-core Qt6::Test)
endif()
