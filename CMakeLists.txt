cmake_minimum_required(VERSION 3.25)
set(HWS_VERSION "0.3.5" CACHE STRING "Application release version")
project(hyprwallselector VERSION ${HWS_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 6.10 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Qml Network Concurrent ShaderTools Test)
if(WIN32)
 find_package(Qt6 REQUIRED COMPONENTS Widgets Multimedia)
else()
 find_package(Qt6 REQUIRED COMPONENTS DBus)
 find_package(LayerShellQt REQUIRED)
endif()
if(NOT WIN32)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
endif()
if(NOT WIN32)
pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
endif()
add_library(hws-core STATIC src/dmspalette.cpp src/dmspalette.h src/controller.cpp src/controller.h src/mediaitem.h src/integration.cpp src/integration.h)
target_include_directories(hws-core PUBLIC src)
qt_add_executable(hyprwallselector src/main.cpp)
target_compile_definitions(hyprwallselector PRIVATE HWS_APP_VERSION="${PROJECT_VERSION}")
target_link_libraries(hyprwallselector PRIVATE hws-core)
target_link_libraries(hws-core PUBLIC Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Concurrent)
if(WIN32)
 target_sources(hws-core PRIVATE src/windowsplatform.cpp src/windowsplatform.h src/windowsstartup.cpp src/windowsstartup.h src/windowslockscreen.cpp src/windowslockscreen.h src/mediawindows.cpp src/mediawindows.h src/nativevideo.cpp src/nativevideo.h src/videodecoder.h)
 target_link_libraries(hws-core PUBLIC user32 dwmapi wtsapi32 psapi avformat avcodec avutil ole32 oleaut32 advapi32 uuid runtimeobject d3d11 Qt6::Multimedia)
 target_link_libraries(hyprwallselector PRIVATE Qt6::Widgets)
 set_target_properties(hyprwallselector PROPERTIES WIN32_EXECUTABLE TRUE)
 configure_file(packaging/windows/app.rc.in app.rc @ONLY)
 target_sources(hyprwallselector PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc src/updatemanager.cpp src/updatemanager.h)
 target_include_directories(hyprwallselector PRIVATE packaging/windows)
else()
 target_sources(hws-core PRIVATE src/mediaitem.cpp)
 target_link_libraries(hws-core PUBLIC PkgConfig::MPV Qt6::DBus LayerShellQt::Interface PkgConfig::SYSTEMD)
endif()
qt_add_resources(hws-core ui PREFIX / FILES packaging/hyprwallselector.svg qml/WindowsVideo.qml qml/Main.qml qml/Card.qml qml/Settings.qml qml/Wallpaper.qml qml/SoftButton.qml qml/SettingsToggle.qml qml/SettingsCombo.qml qml/FolderPicker.qml)
qt_add_shaders(hws-core shaders PREFIX / FILES shaders/card.frag)
if(NOT WIN32)
add_executable(hyprwallselector-hotkeys src/hotkeys.cpp src/chord.h)
target_link_libraries(hyprwallselector-hotkeys PRIVATE Qt6::Core Qt6::DBus PkgConfig::SYSTEMD PkgConfig::UDEV)
target_compile_options(hws-core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(hyprwallselector-hotkeys PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS hyprwallselector RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS hyprwallselector-hotkeys RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
install(FILES packaging/org.hyprwallselector.Hotkeys.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/system.d)
configure_file(packaging/org.hyprwallselector.Hotkeys.service.in org.hyprwallselector.Hotkeys.service @ONLY)
configure_file(packaging/hyprwallselector-hotkeys.service.in hyprwallselector-hotkeys.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.hyprwallselector.Hotkeys.service DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/system-services)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/hyprwallselector-hotkeys.service DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/system)
install(FILES packaging/hyprwallselector.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/hyprwallselector.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/hyprwallselector)
install(FILES docs/SECURITY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/hyprwallselector)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/hyprwallselector)
include(CTest)
if(BUILD_TESTING AND NOT WIN32)
 add_executable(test-chord tests/test-chord.cpp src/chord.h)
 target_include_directories(test-chord PRIVATE src)
 target_link_libraries(test-chord PRIVATE Qt6::Test)
 add_test(NAME chord COMMAND test-chord)
 add_executable(test-ui tests/test-ui.cpp)
 target_link_libraries(test-ui PRIVATE hws-core Qt6::Test)
endif()
if(BUILD_TESTING AND NOT WIN32)
 add_executable(test-integration tests/test-integration.cpp)
 target_link_libraries(test-integration PRIVATE hws-core Qt6::Test)
 add_test(NAME integration COMMAND test-integration)
endif()

if(BUILD_TESTING AND NOT WIN32)
 add_executable(test-palette tests/test-palette.cpp src/dmspalette.cpp src/dmspalette.h)
 target_include_directories(test-palette PRIVATE src)
 target_link_libraries(test-palette PRIVATE Qt6::Test)
 add_test(NAME palette COMMAND test-palette)
endif()

if(WIN32 AND BUILD_TESTING)
 add_executable(test-windows-state tests/test-windows-state.cpp src/windowsdisplays.h)
 target_link_libraries(test-windows-state PRIVATE hws-core Qt6::Test)
 add_test(NAME windows-state COMMAND test-windows-state)
 add_executable(test-windows-ui tests/test-windows-ui.cpp)
 target_link_libraries(test-windows-ui PRIVATE hws-core Qt6::Test)
 add_test(NAME windows-ui COMMAND test-windows-ui)
 add_executable(test-nativevideo tests/test-nativevideo.cpp)
 target_link_libraries(test-nativevideo PRIVATE hws-core Qt6::Test)
 add_test(NAME nativevideo COMMAND test-nativevideo)
 add_executable(test-windows tests/test-windows.cpp)
 target_link_libraries(test-windows PRIVATE hws-core Qt6::Test)
 add_test(NAME windows COMMAND test-windows)
 add_executable(test-updates tests/test-updates.cpp src/updatemanager.cpp src/updatemanager.h)
 target_include_directories(test-updates PRIVATE src)
 target_link_libraries(test-updates PRIVATE Qt6::Test Qt6::Network)
 add_test(NAME updates COMMAND test-updates)
endif()
